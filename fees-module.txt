/* =========================================================
   MASTER SETTINGS SAFE IMPORT
========================================================= */

export const getFeeSettingsData = (
    feeSettings = {}
) => {
    return {
        feeStructure:
            feeSettings?.feeStructure || [],

        lateFee:
            toNumber(feeSettings?.lateFee) || 0,

        enableLateFee:
            feeSettings?.enableLateFee ?? true,

        enableDiscount:
            feeSettings?.enableDiscount ?? true,

        receiptPrefix:
            feeSettings?.receiptPrefix || "RCPT",
    };
};

/* =========================================================
   CLASS FEE STRUCTURE FINDER
========================================================= */

export const getClassFeeStructure = ({
    feeSettings = {},
    className = "",
}) => {
    const { feeStructure } =
        getFeeSettingsData(feeSettings);

    return (
        feeStructure.find(
            (item) => item?.className === className
        ) || null
    );
};

/* =========================================================
   TOTAL FEE CALCULATOR
========================================================= */

export const calculateTotalFee = ({
    feeSettings = {},
    className = "",
}) => {
    const structure = getClassFeeStructure({
        feeSettings,
        className,
    });

    if (!structure) return 0;

    return Object.entries(structure).reduce(
        (total, [key, value]) => {
            if (
                key === "id" ||
                key === "className"
            ) {
                return total;
            }

            return total + toNumber(value);
        },
        0
    );
};

/* =========================================================
   INSTALLMENT SPLITTER
========================================================= */

export const splitInstallments = ({
    totalFee = 0,
    installments = 1,
}) => {
    const total = toNumber(totalFee);
    const count = toNumber(installments);

    if (count <= 1) {
        return [total];
    }

    const baseAmount = Math.floor(total / count);

    const result = Array(count).fill(baseAmount);

    const usedAmount = baseAmount * count;

    const remaining = total - usedAmount;

    result[result.length - 1] += remaining;

    return result;
};

   SMART MONTHLY SPLIT
========================================================= */

export const generateMonthlyInstallments = ({
    totalFee = 0,
}) => {
    return splitInstallments({

        installments: 12,
    });
};

export const generateQuarterlyInstallments = ({

/* =========================================================
   LATE FEE CALCULATOR
========================================================= */

export const calculateLateFee = ({
    dueDate,
    paymentDate = new Date(),
    lateFeePerDay = 0,
    maxLateFee = 5000,
}) => {
    if (!dueDate) return 0;

    const due = new Date(dueDate);
    const paid = new Date(paymentDate);

    if (paid <= due) {
        return 0;
    }

    const diffTime =
        paid.getTime() - due.getTime();

    const daysLate = Math.ceil(
        diffTime / (1000 * 60 * 60 * 24)
    );

    const lateFee =
        daysLate * toNumber(lateFeePerDay);

    return lateFee > maxLateFee
        ? maxLateFee
        : lateFee;
};

/* =========================================================
   DISCOUNT CALCULATOR
========================================================= */

export const calculateDiscount = ({
    amount = 0,
    discountType = "flat",
    discountValue = 0,
}) => {
    const feeAmount = toNumber(amount);
    const value = toNumber(discountValue);

    if (discountType === "percentage") {
        return Math.floor(
            (feeAmount * value) / 100
        );
    }

    return value;
};

/* =========================================================
   FINAL PAYMENT CALCULATOR
========================================================= */

export const calculateFinalPayment = ({
    amount = 0,
    lateFee = 0,
    discount = 0,
}) => {
    const total =
        toNumber(amount) +
        toNumber(lateFee) -
        toNumber(discount);

    return total > 0 ? total : 0;
};

/* =========================================================
   STUDENT FEES RECORD BUILDER
========================================================= */

export const buildStudentFeesRecord = ({
    student = {},
    feeSettings = {},
}) => {
    const totalFee = calculateTotalFee({
        feeSettings,
        className: student?.className,
    });

    const paidAmount = 0;

    const dueAmount = calculateDueAmount({
        totalFee,
        paidAmount,
    });

    return {
        studentId:
            student?.id || student?.studentId,

        admissionNumber:
            student?.admissionNumber || "",

        studentName:
            student?.studentName ||
            student?.name ||
            "",

        className:
            student?.className || "",

        section:
            student?.section || "",

        rollNumber:
            student?.rollNumber || "",

        fatherName:
            student?.fatherName || "",

        mobile:
            student?.mobile || "",

        totalFee,
        paidAmount,
        dueAmount,

        status: PAYMENT_STATUS.UNPAID,

        payments: [],

        createdAt:
            new Date().toISOString(),

        updatedAt:
            new Date().toISOString(),
    };
};

/* =========================================================
   APPLY PAYMENT TO RECORD
========================================================= */

export const applyPaymentToStudent = ({
    studentRecord = {},
    paymentData = {},
}) => {
    const currentPaid = toNumber(
        studentRecord?.paidAmount
    );

    const paymentAmount = calculateFinalPayment({
        amount: paymentData?.amount,
        lateFee: paymentData?.lateFee,
        discount: paymentData?.discount,
    });

    const updatedPaid =
        currentPaid + paymentAmount;

    const updatedDue = calculateDueAmount({
        totalFee: studentRecord?.totalFee,
        paidAmount: updatedPaid,
    });

    const updatedStatus =
        getPaymentStatus({
            totalFee:
                studentRecord?.totalFee,
            paidAmount: updatedPaid,
        });

    return {
        ...studentRecord,

        paidAmount: updatedPaid,

        dueAmount: updatedDue,

        status: updatedStatus,

        payments: [
            ...(studentRecord?.payments ||
                []),
            paymentData,
        ],

        updatedAt:
            new Date().toISOString(),
    };
};

/* =========================================================
   OVERDUE DETECTOR
========================================================= */

export const markOverdueStudents = ({
    feesData = [],
    dueDate,
}) => {
    if (!dueDate) return feesData;

    const today = new Date();
    const lastDate = new Date(dueDate);

    if (today <= lastDate) {
        return feesData;
    }

    return feesData.map((student) => {
        if (
            student?.status ===
            PAYMENT_STATUS.PAID ||
            toNumber(student?.dueAmount) <= 0
        ) {
            return student;
        }

        return {
            ...student,
            status: PAYMENT_STATUS.OVERDUE,
        };
    });
};

/* =========================================================
   COLLECTION SUMMARY
========================================================= */

export const calculateCollectionStats = (
    feesData = []
) => {
    return feesData.reduce(
        (acc, student) => {
            const totalFee = toNumber(
                student?.totalFee
            );

            const paidAmount = toNumber(
                student?.paidAmount
            );

            const dueAmount = toNumber(
                student?.dueAmount
            );

            acc.totalFees += totalFee;
            acc.totalCollection +=
                paidAmount;
            acc.totalDue += dueAmount;

            if (
                student?.status ===
                PAYMENT_STATUS.PAID
            ) {
                acc.paidStudents += 1;
            }

            if (
                student?.status ===
                PAYMENT_STATUS.PARTIAL ||
                student?.status ===
                PAYMENT_STATUS.UNPAID ||
                student?.status ===
                PAYMENT_STATUS.OVERDUE
            ) {
                acc.dueStudents += 1;
            }

            return acc;
        },
        {
            totalFees: 0,
            totalCollection: 0,
            totalDue: 0,
            paidStudents: 0,
            dueStudents: 0,
        }
    );
};

};

            feesData.map((item) =>

        saveFeesDB(finalDB);

        return finalDB;
    };

                String(studentId)

                return updatedStudent;
            }

            return item;
        }

    saveFeesDB(updatedDB);

    return {

import {
    PAYMENT_STATUS,
    RECEIPT_PREFIX,

/* =========================================================
   SAFE NUMBER
========================================================= */

export const toNumber = (value) => {
    const num = Number(value);
    return Number.isNaN(num) ? 0 : num;
};

/* =========================================================
   CURRENCY FORMAT
========================================================= */

export const formatCurrency = (amount = 0) => {
    return `₹${toNumber(amount).toLocaleString("en-IN")}`;
};

/* =========================================================
   DATE FORMAT
========================================================= */

export const formatDate = (dateValue) => {
    if (!dateValue) return "-";

    try {
        const date = new Date(dateValue);

        return date.toLocaleDateString("en-IN", {
            day: "2-digit",
            month: "short",
            year: "numeric",
        });
    } catch (error) {
        return "-";
    }
};

/* =========================================================
   TIME FORMAT
========================================================= */

export const formatTime = (dateValue) => {
    if (!dateValue) return "-";

    try {
        const date = new Date(dateValue);

        return date.toLocaleTimeString("en-IN", {
            hour: "2-digit",
            minute: "2-digit",
        });
    } catch (error) {
        return "-";
    }
};

/* =========================================================
   FULL DATE TIME
========================================================= */

export const formatDateTime = (dateValue) => {
    if (!dateValue) return "-";

    try {
        const date = new Date(dateValue);

        return `${formatDate(date)} • ${formatTime(date)}`;
    } catch (error) {
        return "-";
    }
};

/* =========================================================
   STATUS CALCULATOR
========================================================= */

export const getPaymentStatus = ({
    totalFee = 0,
    paidAmount = 0,
}) => {
    const total = toNumber(totalFee);
    const paid = toNumber(paidAmount);

    if (paid <= 0) {
        return PAYMENT_STATUS.UNPAID;
    }

    if (paid >= total) {
        return PAYMENT_STATUS.PAID;
    }

    return PAYMENT_STATUS.PARTIAL;
};

/* =========================================================
   DUE AMOUNT
========================================================= */

export const calculateDueAmount = ({
    totalFee = 0,
    paidAmount = 0,
}) => {
    const due = toNumber(totalFee) - toNumber(paidAmount);

    return due > 0 ? due : 0;
};

        RECEIPT_PADDING,
        "0"
    )}`;
};

    return Array.isArray(data) ? data : [];
};

/* =========================================================
   SAFE OBJECT
========================================================= */

export const safeObject = (data) => {
    return data && typeof data === "object" ? data : {};
};

/* =========================================================
   SEARCH FILTER
========================================================= */

export const matchesSearch = (student, search = "") => {
    if (!search?.trim()) return true;

    const query = search.toLowerCase();

    return (
        student?.studentName?.toLowerCase()?.includes(query) ||
        student?.className?.toLowerCase()?.includes(query) ||
        student?.section?.toLowerCase()?.includes(query) ||
        String(student?.rollNumber || "")
            .toLowerCase()
            .includes(query) ||
        String(student?.admissionNumber || "")
            .toLowerCase()
            .includes(query)
    );
};

/* =========================================================
   CLASS FILTER
========================================================= */

export const matchesClass = (
    student,
    className = "All"
) => {
    if (className === "All") return true;

    return student?.className === className;
};

/* =========================================================
   SECTION FILTER
========================================================= */

export const matchesSection = (
    student,
    section = "All"
) => {
    if (section === "All") return true;

    return student?.section === section;
};

/* =========================================================
   STATUS FILTER
========================================================= */

export const matchesStatus = (
    student,
    status = "All"
) => {
    if (status === "All") return true;

    return student?.status === status;
};

/* =========================================================
   PROFESSIONAL FILTER ENGINE
========================================================= */

export const filterFeesData = ({
    data = [],
    filters = {},
}) => {
    return safeArray(data).filter((student) => {
        return (
            matchesSearch(student, filters.search) &&
            matchesClass(student, filters.className) &&
            matchesSection(student, filters.section) &&
            matchesStatus(student, filters.status)
        );
    });
};

/* =========================================================
   SORT BY CLASS + ROLL
========================================================= */

export const sortFeesData = (data = []) => {
    return [...safeArray(data)].sort((a, b) => {
        const classCompare = String(a.className).localeCompare(
            String(b.className),
            undefined,
            { numeric: true }
        );

        if (classCompare !== 0) {
            return classCompare;
        }

        return toNumber(a.rollNumber) - toNumber(b.rollNumber);
    });
};

/* =========================================================
   SUMMARY CALCULATOR
========================================================= */

export const calculateSummary = (data = []) => {
    const fees = safeArray(data);

    return fees.reduce(
        (acc, item) => {
            const total = toNumber(item.totalFee);
            const paid = toNumber(item.paidAmount);
            const due = toNumber(item.dueAmount);

            acc.totalCollection += paid;
            acc.totalDue += due;
            acc.totalFees += total;

            if (item.status === PAYMENT_STATUS.PAID) {
                acc.paidStudents += 1;
            }

            if (
                item.status === PAYMENT_STATUS.PARTIAL ||
                item.status === PAYMENT_STATUS.UNPAID ||
                item.status === PAYMENT_STATUS.OVERDUE
            ) {
                acc.dueStudents += 1;
            }

            return acc;
        },
        {
            totalCollection: 0,
            totalDue: 0,
            totalFees: 0,
            paidStudents: 0,
            dueStudents: 0,
        }
    );
};

/* =========================================================
   PAYMENT HISTORY FLATTENER
========================================================= */

export const flattenPaymentHistory = (
    feesData = []
) => {
    const history = [];

    safeArray(feesData).forEach((student) => {
        safeArray(student.payments).forEach((payment) => {
            history.push({
                ...payment,

                studentId: student.studentId,
                studentName: student.studentName,
                className: student.className,
                section: student.section,
                rollNumber: student.rollNumber,

                totalFee: student.totalFee,
                paidAmount: student.paidAmount,
                dueAmount: student.dueAmount,
            });
        });
    });

    return history.sort(
        (a, b) =>
            new Date(b.paymentDate) -
            new Date(a.paymentDate)
    );
};

// src/modules/fees/pages/FeesPage.jsx

import React, { useEffect, useMemo, useState } from "react";
import * as feesService from "../feesService";
import FeesTable from "../components/FeesTable";
import FeesCollectModal from "../components/FeesCollectModal";
import ReceiptModal from "../components/ReceiptModal";
import FeesSummaryCards from "../components/FeesSummaryCards";
import { useSchoolStore } from "../../../store/schoolStore";

const FeesPage = () => {

    const students = useSchoolStore(s => s.students) || [];

    const [search, setSearch] = useState("");
    const [classFilter, setClassFilter] = useState("");

    const [selectedIds, setselectedIds] = useState([]);
    const [activeStudent, setActiveStudent] = useState(null);
    const [receipt, setReceipt] = useState(null);

    /* ================= SYNC ================= */
    useEffect(() => {
        feesService.syncStudentsToFeesDB({ students });
    }, [students]);

    /* ================= FILTER ================= */
    const data = useMemo(() => {
        let list = feesService.getAllFeesRecords();

        if (search) {
            list = list.filter(s =>
                s.studentName?.toLowerCase().includes(search.toLowerCase()) ||
                s.fatherName?.toLowerCase().includes(search.toLowerCase()) ||
                s.mobile?.includes(search)
            );
        }

        if (classFilter) {
            list = list.filter(s => s.className === classFilter);
        }

        return list;
    }, [students, search, classFilter]);

    /* ================= SELECT ================= */
    const toggleSelect = (student) => {
        setselectedIds(prev => {
            const exists = prev.find(s => s.studentId === student.studentId);
            if (exists) {
                return prev.filter(s => s.studentId !== student.studentId);
            }
            return [...prev, student];
        });
    };

    /* ================= LIVE SUMMARY ================= */
    const liveSummary = useMemo(() => {
        if (!selectedIds.length) return null;

        return {
            total: selectedIds.reduce((a, s) => a + Number(s.totalFee || 0), 0),
            paid: selectedIds.reduce((a, s) => a + Number(s.paidAmount || 0), 0),
            due: selectedIds.reduce((a, s) => a + Number(s.dueAmount || 0), 0),
        };
    }, [selectedIds]);

    /* ================= HANDLERS ================= */
    const handleCollect = () => {
        if (selectedIds.length > 0) {
            setActiveStudent(selectedIds[0]);
        }
    };

    const handleSuccess = (res) => {
        setReceipt(res.payment);
        setselectedIds([]);
        setActiveStudent(null);
    };

    /* ================= UI ================= */
    return (
        <div style={{
            padding: "16px",
            background: "#f5f6fa",
            minHeight: "100vh"
        }}>

           
            {/* FILTER BAR (ONLY ONE SEARCH) */}
            <div style={{ display: "flex", gap: "10px", marginBottom: "10px" }}>

                <input
                    placeholder="Search student / father / mobile"
                    value={search}
                    onChange={e => setSearch(e.target.value)}
                />

                <select onChange={e => setClassFilter(e.target.value)}>
                    <option value="">All Class</option>
                    {[...new Set(students.map(s => s.className))].map(c =>
                        <option key={c}>{c}</option>
                    )}
                </select>

                <button onClick={() => {
                    setSearch("");
                    setClassFilter("");
                }}>
                    Reset
                </button>

            </div>

            {/* FIXED + LIVE CARDS */}
            <FeesSummaryCards selected={liveSummary} />

            {/* LIVE ACTION BAR */}
            {selectedIds.length > 0 && (
                <div style={{
                    display: "flex",
                    justifyContent: "space-between",
                    alignItems: "center",
                    background: "#111",
                    color: "#fff",
                    padding: "10px",
                    borderRadius: "12px",
                    marginBottom: "10px"
                }}>
                    <div>
                        Selected Students: {selectedIds.length}
                    </div>

                    <button onClick={handleCollect} style={{
                        padding: "8px 12px",
                        background: "#16a34a",
                        color: "#fff",
                        border: "none",
                        borderRadius: "8px"
                    }}>
                        💰 Collect Fees
                    </button>
                </div>
            )}

            {/* TABLE */}
            <FeesTable
                data={data}
                selected={selectedIds}
                onSelect={toggleSelect}
            />

            {/* MODALS */}
            {activeStudent && (
                <FeesCollectModal
                    student={activeStudent}
                    onClose={() => setActiveStudent(null)}
                    onSuccess={handleSuccess}
                />
            )}

            {receipt && (
                <ReceiptModal
                    receipt={receipt}
                    onClose={() => setReceipt(null)}
                />
            )}

        </div>
    );
};

export default FeesPage;
